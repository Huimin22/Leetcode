```
class Solution:
    def removeInvalidParentheses(self, s: str) -> list[str]:
        res = set()
        
        l_rem, r_rem = 0, 0
        for c in s:
            if c == '(':
                l_rem += 1
            elif c == ')':
                if l_rem > 0:
                    l_rem -= 1
                else:
                    r_rem += 1
        
        def dfs(index, path, l_rem, r_rem, open_count):
            if index == len(s):
                if l_rem == 0 and r_rem == 0 and open_count == 0:
                    res.add(path)
                return
            
            c = s[index]
            
            if c == '(':
                if l_rem > 0:
                    dfs(index + 1, path, l_rem - 1, r_rem, open_count)
                dfs(index + 1, path + '(', l_rem, r_rem, open_count + 1)
            
            elif c == ')':
                if r_rem > 0:
                    dfs(index + 1, path, l_rem, r_rem - 1, open_count)
                if open_count > 0:
                    dfs(index + 1, path + ')', l_rem, r_rem, open_count - 1)
            else:
                dfs(index + 1, path + c, l_rem, r_rem, open_count)
        
        dfs(0, '', l_rem, r_rem, 0)
        return list(res)      

```
