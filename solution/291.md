## 4. Code Implementation:


```
class Solution:
    def wordPatternMatch(self, pattern: str, s: str) -> bool:
        mapping = {}
        used = set()
        
        def dfs(p_idx, s_idx):
            if p_idx == len(pattern) and s_idx == len(s):
                return True
            if p_idx == len(pattern) or s_idx == len(s):
                return False
            
            c = pattern[p_idx]
            if c in mapping:
                mapped_str = mapping[c]
                if not s.startswith(mapped_str, s_idx):
                    return False
                return dfs(p_idx + 1, s_idx + len(mapped_str))
            else:
                for end in range(s_idx + 1, len(s) + 1):
                    candidate = s[s_idx:end]
                    if candidate in used:
                        continue
                    mapping[c] = candidate
                    used.add(candidate)
                    if dfs(p_idx + 1, end):
                        return True
                    del mapping[c]
                    used.remove(candidate)
                return False
        
        return dfs(0, 0)   


```
