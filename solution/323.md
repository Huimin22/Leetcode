```
class Solution:
    def countComponents(self, n: int, edges):
        parent = list(range(n))
        rank = [0] * n

        def find(x):
            if parent[x] != x:
                parent[x] = find(parent[x])  # path compression
            return parent[x]

        def union(x, y):
            px, py = find(x), find(y)
            if px == py:
                return 0  # no merge
            if rank[px] > rank[py]:
                parent[py] = px
            elif rank[px] < rank[py]:
                parent[px] = py
            else:
                parent[py] = px
                rank[px] += 1
            return 1  # merged

        res = n  # initially n components
        for u, v in edges:
            res -= union(u, v)

        return res   

```
