```
class Solution:
    def findOrder(self, numCourses: int, prerequisites: List[List[int]]) -> List[int]:
        taken = 0
        cour_g = defaultdict(list)
        n = numCourses
        in_degree = [0]*n
        res = []
       
        for cour, prev in prerequisites:
            cour_g[prev].append(cour)
            in_degree[cour] += 1

        q = deque()
        for i in range(len(in_degree)):
            if in_degree[i] == 0:
                q.append(i)
                

        while q:
            cur_c = q.popleft()
            res.append(cur_c)
            taken += 1
            for nei in cour_g[cur_c]:
                in_degree[nei] -= 1
                if in_degree[nei] == 0:
                    q.append(nei)

        if taken == n: 
            return res
        else:
            return []   

```
