## 2. Summary

Here’s a concise **mental template** for **LeetCode 542 – 01 Matrix** (multi-source BFS) you can copy-paste:

```python
from collections import deque

def updateMatrix(mat):
    m, n = len(mat), len(mat[0])
    res = [[-1]*n for _ in range(m)]
    queue = deque()

    for i in range(m):
        for j in range(n):
            if mat[i][j] == 0:
                res[i][j] = 0
                queue.append((i,j))
    
    directions = [(0,1),(1,0),(0,-1),(-1,0)]
    while queue:
        i,j = queue.popleft()
        for di,dj in directions:
            ni,nj = i+di,j+dj
            if 0<=ni<m and 0<=nj<n and res[ni][nj]==-1:
                res[ni][nj] = res[i][j]+1
                queue.append((ni,nj))
    return res
```

## 3. Key Points

* Multi-source BFS from all 0s
* Queue stores coordinates `(i,j)`
* Distance = previous cell +1
* Mark visited via `res != -1`
* 4 directions only
* Return `res`

```
class Solution:
    def updateMatrix(self, mat: List[List[int]]) -> List[List[int]]:
        m, n = len(mat), len(mat[0])
        q = deque()
        for i in range(m):
            for j in range(n):
                if mat[i][j] == 0:
                    q.append((i, j))
                else:
                    mat[i][j] = float('inf')
        
        dirs = [(1,0), (-1,0), (0,1), (0,-1)]
        
        while q:
            x, y = q.popleft()
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if 0 <= nx < m and 0 <= ny < n:
                    if mat[nx][ny] > mat[x][y] + 1:
                        mat[nx][ny] = mat[x][y] + 1
                        q.append((nx, ny))
        
        return mat   


```
