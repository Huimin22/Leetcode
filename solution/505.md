```
def shortestDistance(maze, start, destination):
    m, n = len(maze), len(maze[0])
    dist = [[float('inf')] * n for _ in range(m)]
    dist[start[0]][start[1]] = 0
    
    q = deque([start])
    dirs = [(0,1),(0,-1),(1,0),(-1,0)]
    
    while q:
        x, y = q.popleft()
        for dx, dy in dirs:
            nx, ny, d = x, y, dist[x][y]

           #hit the wall and stop
            while 0 <= nx+dx < m and 0 <= ny+dy < n and maze[nx+dx][ny+dy] == 0:
                nx += dx
                ny += dy
                d += 1

            # find shorter distance and update dist
            if d < dist[nx][ny]:
                dist[nx][ny] = d
                q.append((nx, ny))
                
    return dist[destination[0]][destination[1]] if dist[destination[0]][destination[1]] != float('inf') else -1   
```
