## 4. Code Implementation:

```
class Solution:
    def numIslands2(self, m: int, n: int, positions: list[list[int]]) -> list[int]:
        parent = {}
        count = 0
        res = []

        def find(x):
            parent.setdefault(x, x)
            if parent[x] != x:
                parent[x] = find(parent[x])
            return parent[x]

        def union(x, y):
            px, py = find(x), find(y)
            if px != py:
                parent[px] = py
                return True
            return False

        for r, c in positions:
            idx = r * n + c
            if idx in parent:
                res.append(count)
                continue
            parent[idx] = idx
            count += 1

            for dr, dc in [(0,1),(0,-1),(1,0),(-1,0)]:
                nr, nc = r + dr, c + dc
                nidx = nr * n + nc
                if 0 <= nr < m and 0 <= nc < n and nidx in parent:
                    if union(idx, nidx):
                        count -= 1
            res.append(count)

        return res   

```
