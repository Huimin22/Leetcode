## 4. Code Implementation:


```
class Solution:
    def partition(self, s: str) -> list[list[str]]:
        res = []
        def is_palindrome(sub: str) -> bool:
            return sub == sub[::-1]
        def dfs(start: int, path: list[str]):
            if start == len(s):
                res.append(path[:])
                return
            for end in range(start + 1, len(s) + 1):
                sub = s[start:end]
                if is_palindrome(sub):
                    dfs(end, path + [sub])
        dfs(0, [])
        return res   

```
