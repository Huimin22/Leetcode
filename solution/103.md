queue = deque([root])
flag = True  # left->right

while queue:
    level = deque()
    for _ in range(len(queue)):
        node = queue.popleft()
        if flag:

## 3. Key Points

## 4. Code Implementation:

```
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []
        q = deque([root])
        flag = True
        res = []
        while q: 
            n = len(q)
            level = deque()

            for _ in range(n):
                cur_n = q.popleft()
                if flag:
                    level.append(cur_n.val)
                else:
                    level.appendleft(cur_n.val)

                if cur_n.left:
                    q.append(cur_n.left)
                if cur_n.right:
                    q.append(cur_n.right)
            res.append(list(level))
            flag = not flag
        return res   
```
